VPC를 aws resource 단위로 직접 생성 시 코드
```sh
resource "aws_vpc" "my" {
    cidr_block  = "172.16.0.0/16"

    tags = {
        Name = "my-vpc"
    }
}
```
```sh
resource "aws_subnet" "my-pub-sb1" {
    vpc_id            = aws_vpc.my.id
    cidr_block        = "172.16.10.0/24"

    availability_zone = "ap-northeast-2a"
    map_public_ip_on_launch = true
    tags = {
        Name = "my-pub-sb1"
    }
}
```
```sh
resource "aws_subnet" "my-pub-sb2" {
    vpc_id            = aws_vpc.my.id
    cidr_block        = "172.16.20.0/24"

    availability_zone = "ap-northeast-2c"
    map_public_ip_on_launch = true
    tags = {
        Name = "my-pub-sb2"
    }
}
```
```sh
resource "aws_subnet" "my-pri-sb1" {
    vpc_id            = aws_vpc.my.id
    cidr_block        = "172.16.110.0/24"

    availability_zone = "ap-northeast-2a"

    tags = {
        Name = "my-pri-sb1"
    }
}
```
```sh
resource "aws_subnet" "my-pri-sb2" {
    vpc_id            = aws_vpc.my.id
    cidr_block        = "172.16.120.0/24"

    availability_zone = "ap-northeast-2c"

    tags = {
        Name = "my-pri-sb2"
    }
}
```
```sh
resource "aws_internet_gateway" "my-igw" {
    vpc_id = aws_vpc.my.id

    tags={
        Name= "my-igw"
    }
}
``` 
```sh
resource "aws_route_table" "my-pub-rtb" {
    vpc_id = aws_vpc.my.id

    tags={
        Name = "my-pub-rtb"
    }
}
```
```sh
resource "aws_route" "my-pub-rtb-conf" {
    route_table_id      = aws_route_table.my-pub-rtb.id
    destination_cidr_block = "0.0.0.0/0"
    gateway_id             = aws_internet_gateway.my-igw.id
}
```
```sh
resource "aws_route_table_association" "route_table_association1" {
    subnet_id      = aws_subnet.my-pub-sb1.id
    route_table_id = aws_route_table.my-pub-rtb.id
}
```
```sh
resource "aws_route_table_association" "route_table_association2" {
    subnet_id      = aws_subnet.my-pub-sb2.id
    route_table_id = aws_route_table.my-pub-rtb.id
}
```

VPC 모듈 사용 시 코드
```sh
module "vpc" {
  source = "terraform-aws-modules/vpc/aws"

  name = "my-vpc"
  cidr = "172.16.0.0/16"

  azs             = ["ap-northeast-2a", "ap-northeast-2b"]
  private_subnets = ["172.16.110.0/24", "172.16.120.0/24"]
  public_subnets  = ["172.16.10.0/24", "172.16.20.0/24"]

  enable_nat_gateway = false
  enable_vpn_gateway = false

  tags = {
    Terraform = "true"
    Environment = "dev"
  }
}
```
