Count 사용 예시 

1.단순 횟수 반복
```sh
resource "aws_instance" "server" {
  count = 4 

  ami           = "ami-a1b2c3d4"
  instance_type = "t2.micro"
}
```

2. 단순 횟수 반복 및 숫자 지정

```sh
resource "aws_instance" "server" {
  count = 4 

  ami           = "ami-a1b2c3d4"
  instance_type = "t2.micro"

  tags = {
    Name = "Server ${count.index}"
  }
}
```

for_each 반복 구문 예시

1. 반복문만 사용

```sh
resource "aws_instance" "web" {
  for_each = {
    "web-1" = "ap-northeast-2a"
    "web-2" = "ap-northeast-2b"
    "web-3" = "ap-northeast-2c"
  }

  ami               = "ami-12345678"
  instance_type     = "t3.micro"
  availability_zone = each.value

  tags = {
    Name = each.key
  }
}
```


2. 변수 포함 사용
```sh
variable "region" {
  default = "ap-northeast-2"
}

locals {
  instances = {
    "web-1" = "${var.region}a"
    "web-2" = "${var.region}b"
    "web-3" = "${var.region}c"
  }
}
```

```sh
resource "aws_instance" "web" {
  for_each = local.instances

  ami           = "ami-a1b2c3d4"
  instance_type = "t3.micro"
  availability_zone = each.value

  tags = {
    Name = each.key
  }
}
```
